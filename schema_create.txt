-- 创建换热站业务信息表
CREATE TABLE ods_sc01_ntjt_zhgr_heat_station_info (
    -- 主键和标识字段
    id INTEGER AUTO_INCREMENT PRIMARY KEY COMMENT '自增整数类型，用于唯一标识记录',
    stn_id INTEGER NULL COMMENT '实时表中对应的换热站id',
    heatstation_code VARCHAR(50) NULL COMMENT 'ERP中换热站唯一编码',
    
    -- 基本信息字段
    stn_name VARCHAR(200) NULL COMMENT '换热站名称',
    stn_state CHAR(1) NULL COMMENT '换热站状态（0：在运，1：停运，2：维修）',
    plan_num INTEGER NULL COMMENT '机组数量',
    heat_method CHAR(1) NULL COMMENT '采暖方式(0:地暖/1:挂暖/2:中央空调/3:挂暖+中央空调)',
    build_type CHAR(1) NULL COMMENT '站点性质(0:居民/1:公建/2:医院/3:学校/4:工业)',
    site_type CHAR(1) NULL COMMENT '站点类型（0：监控站，1：人工站，2：自管站）',
    source_type CHAR(1) NULL COMMENT '热源形式(0:地热井/风冷螺杆机/1:空气源热泵/2:燃气锅炉/3:燃空气源热泵/4:水网/5:基汽/6:综合能源)',
    energy_save_type CHAR(1) NULL COMMENT '节能类型(0:非节能/1:一步节能/2:二步节能/3:三步节能/4:四步节能)',
    
    -- 面积相关字段
    heat_area NUMERIC(15,2) NULL COMMENT '供热面积（默认）',
    mesh_area NUMERIC(15,2) NULL COMMENT '入网面积(建筑面积)',
    trans_area NUMERIC(15,2) NULL COMMENT '传热面积',
    actual_area NUMERIC(15,2) NULL COMMENT '实际供暖面积',
    
    -- 时间字段
    state_time TIMESTAMP NULL COMMENT '状态变更时间',
    build_time TIMESTAMP NULL COMMENT '建设时间',
    create_time TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time TIMESTAMP NULL COMMENT '更新时间',
    
    -- 位置和联系信息
    longitude NUMERIC(10,6) NULL COMMENT '经度',
    latitude NUMERIC(10,6) NULL COMMENT '纬度',
    distance NUMERIC(15,2) NULL COMMENT '距离（拓扑距离）真实管网距离',
    site_addr VARCHAR(500) NULL COMMENT '热站地址',
    belong_unit VARCHAR(200) NULL COMMENT '所属单位',
    charge_person VARCHAR(100) NULL COMMENT '负责人',
    contact_info VARCHAR(100) NULL COMMENT '联系人',
    
    -- 设备数量
    source_num INTEGER NULL COMMENT '热源数量',
    valve_num INTEGER NULL COMMENT '阀门数量',
    
    -- 系统字段
    create_by VARCHAR(100) NULL COMMENT '创建人',
    update_by VARCHAR(100) NULL COMMENT '更新人',
    is_delete CHAR(1) DEFAULT '0' COMMENT '0：未删除 1：已删除',
    remark VARCHAR(500) NULL COMMENT '备注',
    serial_num INTEGER NULL COMMENT '站点排序',
    
    -- 索引
    INDEX idx_stn_id (stn_id),
    INDEX idx_heatstation_code (heatstation_code),
    INDEX idx_stn_state (stn_state),
    INDEX idx_belong_unit (belong_unit)
) ENGINE=InnoDB 
  DEFAULT CHARSET=utf8mb4 
  COLLATE=utf8mb4_general_ci
